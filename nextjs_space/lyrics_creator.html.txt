(function shibZLyricsCreator(){
"use strict";

// ── shibZ Lyrics Creator – State ──
let audioCtx = null;
let audioBuffer = null;
let sourceNode = null;
let isPlaying = false;
let startedAt = 0;   // audioCtx.currentTime when playback started
let pausedAt = 0;    // offset in seconds where we paused
let taps = [];       // array of seconds
let manualBreaks = []; // array of seconds (manual phrase boundaries)
let gapThreshold = 1.5;
let waveformData = null; // Float32Array of peaks
let animFrame = null;

// ── DOM refs ──
const uploadZone = document.getElementById('upload-zone');
const uploadSection = document.getElementById('upload-section');
const fileInput = document.getElementById('file-input');
const playerSection = document.getElementById('player-section');
const promptSection = document.getElementById('prompt-section');
const fileNameEl = document.getElementById('file-name');
const tapBadge = document.getElementById('tap-badge');
const timeDisplay = document.getElementById('time-display');
const canvas = document.getElementById('waveform-canvas');
const ctx = canvas.getContext('2d');
const tooltip = document.getElementById('tooltip');
const rewindBanner = document.getElementById('rewind-banner');
const gapSlider = document.getElementById('gap-slider');
const gapVal = document.getElementById('gap-val');
const phraseTbody = document.getElementById('phrase-tbody');
const langInput = document.getElementById('lang-input');
const styleInput = document.getElementById('style-input');
const extraInput = document.getElementById('extra-input');
const copyBtn = document.getElementById('copy-prompt-btn');
const copyConfirm = document.getElementById('copy-confirm');
const promptOutput = document.getElementById('prompt-output');
const waveformWrap = document.getElementById('waveform-wrap');

// ── Phrase colors ──
const PHRASE_COLORS = ['#7c6cff','#ff6ca8','#4ade80','#ffaa44','#60b5ff','#ff9898','#a19ad3','#80d8c3','#ff5555','#f0e060'];

// ── Helpers ──
function shibZFmtTime(s){
  if(!isFinite(s)||s<0) s=0;
  const m=Math.floor(s/60);
  const sec=s-m*60;
  return m+':'+sec.toFixed(3).padStart(6,'0');
}
function shibZFmtShort(s){
  if(!isFinite(s)||s<0) s=0;
  const m=Math.floor(s/60);
  const sec=Math.floor(s-m*60);
  return m+':'+String(sec).padStart(2,'0');
}
function shibZCurrentTime(){
  if(!audioCtx||!audioBuffer) return 0;
  if(!isPlaying) return pausedAt;
  const t = audioCtx.currentTime - startedAt + pausedAt;
  return Math.min(t, audioBuffer.duration);
}

// ── Audio init ──
function shibZInitAudioCtx(){
  if(!audioCtx) audioCtx = new (window.AudioContext||window.webkitAudioContext)();
}

// ── File handling ──
uploadZone.addEventListener('dragover', e=>{e.preventDefault();uploadZone.classList.add('dragover')});
uploadZone.addEventListener('dragleave', ()=>uploadZone.classList.remove('dragover'));
uploadZone.addEventListener('drop', e=>{
  e.preventDefault();uploadZone.classList.remove('dragover');
  const f=e.dataTransfer?.files?.[0];
  if(f) shibZLoadFile(f);
});
fileInput.addEventListener('change', ()=>{
  const f=fileInput?.files?.[0];
  if(f) shibZLoadFile(f);
});

async function shibZLoadFile(file){
  shibZInitAudioCtx();
  if(audioCtx.state==='suspended') await audioCtx.resume();
  shibZStopPlayback();
  taps=[];
  manualBreaks=[];
  const buf = await file.arrayBuffer();
  audioBuffer = await audioCtx.decodeAudioData(buf);
  fileNameEl.textContent = file.name;
  uploadSection.style.display='none';
  playerSection.style.display='block';
  promptSection.style.display='block';
  shibZComputeWaveform();
  shibZStartRenderLoop();
  shibZUpdatePhraseTable();
}

// ── Waveform computation ──
function shibZComputeWaveform(){
  const raw = audioBuffer.getChannelData(0);
  const w = canvas.clientWidth;
  const dpr = window.devicePixelRatio||1;
  canvas.width = w*dpr;
  canvas.height = canvas.clientHeight*dpr;
  ctx.setTransform(dpr,0,0,dpr,0,0);
  const samplesPerPx = Math.floor(raw.length / w);
  const peaks = new Float32Array(w);
  for(let i=0;i<w;i++){
    let max=0;
    const start=i*samplesPerPx;
    for(let j=start;j<start+samplesPerPx&&j<raw.length;j++){
      const v=Math.abs(raw[j]);
      if(v>max) max=v;
    }
    peaks[i]=max;
  }
  waveformData=peaks;
}

// ── Playback ──
function shibZPlay(){
  if(!audioBuffer) return;
  shibZInitAudioCtx();
  if(audioCtx.state==='suspended') audioCtx.resume();
  if(sourceNode){try{sourceNode.stop()}catch(e){}}
  sourceNode = audioCtx.createBufferSource();
  sourceNode.buffer = audioBuffer;
  sourceNode.connect(audioCtx.destination);
  sourceNode.onended = ()=>{
    if(isPlaying){
      isPlaying=false;
      pausedAt=audioBuffer.duration;
    }
  };
  startedAt = audioCtx.currentTime;
  sourceNode.start(0, pausedAt);
  isPlaying=true;
  rewindBanner.style.display='none';
}
function shibZPause(){
  if(!isPlaying) return;
  pausedAt = shibZCurrentTime();
  if(sourceNode){try{sourceNode.stop()}catch(e){}}
  isPlaying=false;
}
function shibZStopPlayback(){
  shibZPause();
  pausedAt=0;
}
function shibZSeekTo(t){
  const wasPlaying=isPlaying;
  if(isPlaying) shibZPause();
  pausedAt=Math.max(0,Math.min(t, audioBuffer?.duration??0));
  if(wasPlaying) shibZPlay();
}

// ── Phrase detection ──
function shibZDetectPhrases(){
  const sorted = [...taps].sort((a,b)=>a-b);
  // Merge manual breaks
  const allBreaks = new Set(manualBreaks);
  const phrases=[];
  let cur=[];
  for(let i=0;i<sorted.length;i++){
    if(cur.length>0){
      const gap = sorted[i]-sorted[i-1];
      // check if a manual break exists between sorted[i-1] and sorted[i]
      let hasManual = false;
      for(const mb of allBreaks){
        if(mb>sorted[i-1]&&mb<sorted[i]){hasManual=true;break;}
      }
      if(gap>gapThreshold||hasManual){
        phrases.push([...cur]);
        cur=[];
      }
    }
    cur.push(sorted[i]);
  }
  if(cur.length) phrases.push(cur);
  return phrases;
}

function shibZUpdatePhraseTable(){
  const phrases = shibZDetectPhrases();
  phraseTbody.innerHTML='';
  if(phrases.length===0){
    phraseTbody.innerHTML='<tr><td colspan="5" style="color:var(--text-dim);text-align:center">No taps recorded yet</td></tr>';
    return;
  }
  phrases.forEach((ph,i)=>{
    const color = PHRASE_COLORS[i%PHRASE_COLORS.length];
    const start=ph[0], end=ph[ph.length-1];
    const tr=document.createElement('tr');
    tr.innerHTML=`<td><span class="color-dot" style="background:${color}"></span>${i+1}</td><td>${ph.length}</td><td>${shibZFmtTime(start)}</td><td>${shibZFmtTime(end)}</td><td>${(end-start).toFixed(2)}s</td>`;
    phraseTbody.appendChild(tr);
  });
}

// ── Canvas rendering ──
function shibZDrawFrame(){
  if(!waveformData) return;
  const W=canvas.clientWidth, H=canvas.clientHeight;
  ctx.clearRect(0,0,W,H);
  const dur=audioBuffer?.duration??1;
  const t=shibZCurrentTime();

  // Phrases for coloring
  const phrases=shibZDetectPhrases();
  const tapPhrase=new Map();
  phrases.forEach((ph,pi)=>ph.forEach(tap=>tapPhrase.set(tap,pi)));

  // Draw waveform bars
  for(let i=0;i<waveformData.length;i++){
    const x=i;
    const h=waveformData[i]*H*0.85;
    const barTime=(i/W)*dur;
    // Color: find which phrase this x is in
    let color='#333344';
    // If before needle, slightly brighter
    if(barTime<=t) color='#444460';
    ctx.fillStyle=color;
    ctx.fillRect(x, (H-h)/2, 1, h||1);
  }

  // Draw phrase background bands (subtle)
  phrases.forEach((ph,pi)=>{
    if(ph.length===0) return;
    const x1=(ph[0]/dur)*W;
    const x2=(ph[ph.length-1]/dur)*W;
    const color=PHRASE_COLORS[pi%PHRASE_COLORS.length];
    ctx.fillStyle=color+'15';
    ctx.fillRect(x1,0,x2-x1+1,H);
  });

  // Draw tap markers
  const sorted=[...taps].sort((a,b)=>a-b);
  sorted.forEach((tap)=>{
    const x=(tap/dur)*W;
    const pi=tapPhrase.get(tap)??0;
    const color=PHRASE_COLORS[pi%PHRASE_COLORS.length];
    ctx.strokeStyle=color;
    ctx.lineWidth=1.5;
    ctx.beginPath();
    ctx.moveTo(x,8);
    ctx.lineTo(x,H-8);
    ctx.stroke();
  });

  // Draw manual break markers
  manualBreaks.forEach(mb=>{
    const x=(mb/dur)*W;
    ctx.strokeStyle='#ffaa44';
    ctx.lineWidth=2;
    ctx.setLineDash([4,3]);
    ctx.beginPath();
    ctx.moveTo(x,0);
    ctx.lineTo(x,H);
    ctx.stroke();
    ctx.setLineDash([]);
    // small diamond at top
    ctx.fillStyle='#ffaa44';
    ctx.beginPath();
    ctx.moveTo(x,2);ctx.lineTo(x+5,8);ctx.lineTo(x,14);ctx.lineTo(x-5,8);ctx.closePath();ctx.fill();
  });

  // Draw phrase boundary separators (auto-detected gaps)
  for(let i=1;i<phrases.length;i++){
    const prevEnd=phrases[i-1][phrases[i-1].length-1];
    const nextStart=phrases[i][0];
    const midTime=(prevEnd+nextStart)/2;
    // Only draw if not a manual break nearby
    let isManual=false;
    for(const mb of manualBreaks){
      if(mb>prevEnd&&mb<nextStart){isManual=true;break;}
    }
    if(!isManual){
      const x=(midTime/dur)*W;
      ctx.strokeStyle='#ffffff22';
      ctx.lineWidth=1;
      ctx.setLineDash([2,4]);
      ctx.beginPath();ctx.moveTo(x,0);ctx.lineTo(x,H);ctx.stroke();
      ctx.setLineDash([]);
    }
  }

  // Needle
  const nx=(t/dur)*W;
  ctx.strokeStyle='#fff';
  ctx.lineWidth=2;
  ctx.shadowColor='#fff';
  ctx.shadowBlur=6;
  ctx.beginPath();ctx.moveTo(nx,0);ctx.lineTo(nx,H);ctx.stroke();
  ctx.shadowBlur=0;
  // Needle head
  ctx.fillStyle='#fff';
  ctx.beginPath();ctx.arc(nx,H/2,4,0,Math.PI*2);ctx.fill();

  // Time
  timeDisplay.textContent=shibZFmtTime(t)+' / '+shibZFmtShort(dur);

  // Auto-stop at end
  if(isPlaying && t>=dur){
    shibZPause();
    pausedAt=dur;
  }

  animFrame=requestAnimationFrame(shibZDrawFrame);
}
function shibZStartRenderLoop(){
  if(animFrame) cancelAnimationFrame(animFrame);
  shibZDrawFrame();
}

// ── Canvas interactions ──
// Seek on click
canvas.addEventListener('click', e=>{
  if(!audioBuffer) return;
  const rect=canvas.getBoundingClientRect();
  const x=e.clientX-rect.left;
  const ratio=x/rect.width;
  const t=ratio*audioBuffer.duration;

  // Check if clicking near a manual break to remove it
  const dur=audioBuffer.duration;
  for(let i=0;i<manualBreaks.length;i++){
    const bx=(manualBreaks[i]/dur)*rect.width;
    if(Math.abs(x-bx)<6){
      manualBreaks.splice(i,1);
      shibZUpdatePhraseTable();
      return;
    }
  }

  // Check if clicking between two taps to add a manual break
  const sorted=[...taps].sort((a,b)=>a-b);
  for(let i=0;i<sorted.length-1;i++){
    const tx1=(sorted[i]/dur)*rect.width;
    const tx2=(sorted[i+1]/dur)*rect.width;
    if(x>tx1+6 && x<tx2-6){
      // Add manual break at this time
      manualBreaks.push(t);
      manualBreaks.sort((a,b)=>a-b);
      shibZUpdatePhraseTable();
      return;
    }
  }

  // Otherwise seek
  shibZSeekTo(t);
});

// Tooltip on hover
canvas.addEventListener('mousemove', e=>{
  if(!audioBuffer||!taps.length){
    tooltip.style.display='none';
    return;
  }
  const rect=canvas.getBoundingClientRect();
  const x=e.clientX-rect.left;
  const dur=audioBuffer.duration;
  const hoverTime=(x/rect.width)*dur;

  // Find closest tap
  const sorted=[...taps].sort((a,b)=>a-b);
  let closest=-1, minDist=Infinity;
  sorted.forEach((tap,i)=>{
    const d=Math.abs(tap-hoverTime);
    if(d<minDist){minDist=d;closest=i;}
  });

  const tapPx=(sorted[closest]/dur)*rect.width;
  if(Math.abs(x-tapPx)<12){
    // Determine phrase
    const phrases=shibZDetectPhrases();
    let phraseIdx=0;
    for(let p=0;p<phrases.length;p++){
      if(phrases[p].includes(sorted[closest])){phraseIdx=p+1;break;}
    }
    tooltip.textContent=`Syllable ${closest+1} · ${shibZFmtTime(sorted[closest])} · Phrase ${phraseIdx}`;
    tooltip.style.display='block';
    tooltip.style.left=(tapPx)+'px';
    tooltip.style.top='-28px';
  } else {
    // Check if near a manual break
    let nearBreak=false;
    for(const mb of manualBreaks){
      const bpx=(mb/dur)*rect.width;
      if(Math.abs(x-bpx)<8){
        tooltip.textContent='Click to remove phrase break';
        tooltip.style.display='block';
        tooltip.style.left=bpx+'px';
        tooltip.style.top='-28px';
        nearBreak=true;
        break;
      }
    }
    if(!nearBreak){
      // Check if between two taps
      let betweenTaps=false;
      for(let i=0;i<sorted.length-1;i++){
        const tx1=(sorted[i]/dur)*rect.width;
        const tx2=(sorted[i+1]/dur)*rect.width;
        if(x>tx1+6&&x<tx2-6){
          tooltip.textContent='Click to add phrase break';
          tooltip.style.display='block';
          tooltip.style.left=x+'px';
          tooltip.style.top='-28px';
          betweenTaps=true;
          break;
        }
      }
      if(!betweenTaps) tooltip.style.display='none';
    }
  }
});
canvas.addEventListener('mouseleave',()=>{tooltip.style.display='none'});

// ── Ripple effect ──
function shibZSpawnRipple(){
  if(!audioBuffer) return;
  const dur=audioBuffer.duration;
  const t=shibZCurrentTime();
  const wrap=waveformWrap;
  const rect=wrap.getBoundingClientRect();
  const nx=(t/dur)*rect.width;
  const el=document.createElement('div');
  el.className='ripple';
  el.style.left=(nx-12)+'px';
  el.style.top=(rect.height/2-12)+'px';
  wrap.appendChild(el);
  setTimeout(()=>el.remove(),500);
}

// ── Keyboard ──
document.addEventListener('keydown', e=>{
  // Ignore if typing in inputs
  const tag=e.target?.tagName?.toLowerCase?.()??'';
  if(tag==='input'||tag==='textarea'||tag==='select') return;

  if(e.code==='Space'){
    e.preventDefault();
    if(!audioBuffer) return;
    if(isPlaying) shibZPause(); else shibZPlay();
    return;
  }

  if(e.code==='Backspace'){
    e.preventDefault();
    if(taps.length===0) return;
    const removed=taps.pop();
    shibZPause();
    // Rewind to (last remaining tap - 7s) or 0
    let rewindTo=0;
    if(taps.length>0){
      const lastTap=Math.max(...taps);
      rewindTo=Math.max(0, lastTap-7);
    }
    pausedAt=rewindTo;
    tapBadge.textContent='🎵 '+taps.length+' taps';
    rewindBanner.style.display='block';
    shibZUpdatePhraseTable();
    return;
  }

  // Any other key = record tap
  if(!isPlaying||!audioBuffer) return;
  const t=shibZCurrentTime();
  taps.push(t);
  tapBadge.textContent='🎵 '+taps.length+' taps';
  shibZSpawnRipple();
  shibZUpdatePhraseTable();
});

// ── Gap slider ──
gapSlider.addEventListener('input',()=>{
  gapThreshold=parseFloat(gapSlider.value);
  gapVal.textContent=gapThreshold.toFixed(1)+'s';
  shibZUpdatePhraseTable();
});

// ── Prompt generation ──
copyBtn.addEventListener('click',()=>{
  const phrases=shibZDetectPhrases();
  if(phrases.length===0){
    promptOutput.value='No taps recorded yet — play the track and tap some syllables first!';
    return;
  }
  const lang=(langInput.value||'').trim()||'English';
  const style=(styleInput.value||'').trim();
  const extra=(extraInput.value||'').trim();

  let lines='';
  phrases.forEach((ph,i)=>{
    lines+=`Line ${i+1}: ${ph.length} syllable${ph.length!==1?'s':''}\n`;
  });

  let prompt=`You are a lyricist. Write song lyrics matching the EXACT syllable structure below.\n\n`;
  prompt+=`Language: ${lang}\n`;
  if(style) prompt+=`Style / Mood: ${style}\n`;
  prompt+=`\nSyllable structure:\n${lines}`;
  prompt+=`\nRules:\n- Match the syllable count of each line EXACTLY.\n- Maintain natural rhythm and rhyme.\n- Output ONLY the lyrics, nothing else.\n`;
  if(extra) prompt+=`\nAdditional instructions:\n${extra}\n`;

  promptOutput.value=prompt;

  navigator.clipboard?.writeText?.(prompt)?.then?.(()=>{
    copyConfirm.style.display='inline';
    setTimeout(()=>{copyConfirm.style.display='none'},2000);
  })?.catch?.(()=>{});
});

  if(audioBuffer) shibZComputeWaveform();
});
